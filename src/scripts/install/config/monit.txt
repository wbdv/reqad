check process crond with pidfile /var/run/crond.pid 
    start program = "/usr/bin/systemctl start crond"
    stop program  = "/usr/bin/systemctl stop crond"
    if 3 restarts within 5 cycles then timeout

check process reqad with pidfile /run/reqad.pid
    start program = "/usr/bin/systemctl start reqad"
    stop program  = "/usr/bin/systemctl stop reqad"
    if failed host localhost port 2087 protocol https for 2 cycles then restart
    if 3 restarts within 5 cycles then timeout

check process reqad-php-fpm with pidfile /run/reqad-php-fpm.pid 
    start program = "/usr/bin/systemctl start reqad-php-fpm" with timeout 60 seconds
    stop program  = "/usr/bin/systemctl stop reqad-php-fpm"
    if failed unixsocket /run/reqad-php-fpm.sock then restart
    if 3 restarts within 5 cycles then timeout

    if loadavg (1min) per core > 2 for 5 cycles then alert

    if memory usage > 95% then alert
    if swap usage > 25% then alert
